DEFINE
    VAR _txt = "CODIGOS_AQUI"
    VAR _path = SUBSTITUTE ( SUBSTITUTE ( _txt, " ", "" ), ",", "|" )
    VAR _codes = SELECTCOLUMNS ( GENERATESERIES ( 1, PATHLENGTH ( _path ) ), "c", PATHITEM ( _path, [Value] ) )
    VAR _it = FILTER ( f_items, f_items[barcode] IN _codes )
    VAR q = UNICHAR ( 34 )
    VAR _lista =
        CONCATENATEX (
            _it,
            VAR _i = f_items[f_holdings_script.instanceId]
            VAR _ins = FILTER ( f_instances, f_instances[id] = _i )
            VAR _c = MAXX ( _ins, f_instances[contributors] )
            VAR _p = SEARCH ( "'name': '", _c, 1, 0 )
            VAR _autor = IF ( _p = 0, "", MID ( _c, _p + 9, SEARCH ( "'", _c, _p + 9, LEN ( _c ) + 1 ) - _p - 9 ) )
            VAR _tit = MAXX ( _ins, f_instances[title] )
            VAR _idi = SUBSTITUTE ( SUBSTITUTE ( SUBSTITUTE ( MAXX ( _ins, f_instances[languages] ), "[", "" ), "]", "" ), "'", "" )
            VAR _todos = FILTER ( f_items, f_items[f_holdings_script.instanceId] = _i )
            RETURN
                "{" & q & "codigo" & q & ":" & q & f_items[barcode] & q
                & "," & q & "hrid" & q & ":" & q & f_items[f_instances_script.hrid] & q
                & "," & q & "copia" & q & ":" & q & SUBSTITUTE ( SUBSTITUTE ( f_items[copyNumber], "\", "/" ), q, "'" ) & q
                & "," & q & "tipo" & q & ":" & q & f_items[f_mtypes_script.name] & q
                & "," & q & "ubicacion" & q & ":" & q & TRIM ( f_items[f_locations_script.biblioteca] ) & " " & TRIM ( f_items[f_locations_script.ubicación] ) & q
                & "," & q & "titulo" & q & ":" & q & SUBSTITUTE ( SUBSTITUTE ( SUBSTITUTE ( SUBSTITUTE ( _tit, "\", "/" ), q, "'" ), UNICHAR ( 10 ), " " ), UNICHAR ( 13 ), " " ) & q
                & "," & q & "autor" & q & ":" & q & SUBSTITUTE ( SUBSTITUTE ( _autor, "\", "/" ), q, "'" ) & q
                & "," & q & "idioma" & q & ":" & q & _idi & q
                & "," & q & "fecha" & q & ":" & q & FORMAT ( f_items[CreationDate], "dd/mm/yyyy" ) & q
                & "," & q & "anio" & q & ":" & q & f_items[f_intancesPublication.dateOfPublication] & q
                & "," & q & "exA" & q & ":" & q & ( COUNTROWS ( FILTER ( _todos, CONTAINSSTRING ( f_items[f_locations_script.biblioteca], "Edif. A" ) ) ) + 0 ) & q
                & "," & q & "exF" & q & ":" & q & ( COUNTROWS ( FILTER ( _todos, CONTAINSSTRING ( f_items[f_locations_script.biblioteca], "Edif. F" ) ) ) + 0 ) & q
                & "," & q & "exP" & q & ":" & q & ( COUNTROWS ( FILTER ( _todos, CONTAINSSTRING ( f_items[f_locations_script.biblioteca], "Postgrado" ) ) ) + 0 ) & q
                & "," & q & "exV" & q & ":" & q & ( COUNTROWS ( FILTER ( _todos, CONTAINSSTRING ( f_items[f_locations_script.biblioteca], "Viña" ) ) ) + 0 ) & q
                & "}",
            ","
        )
EVALUATE
ROW ( "datos", "[" & _lista & "]" )
